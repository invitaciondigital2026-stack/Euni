admin_mult.html ya llama al mismo API_URL de admin2 con estas acciones:
- action=media_list

Agregar estos helpers al Apps Script existente. En Configuracion del proyecto > Propiedades del script crear:

- CLOUDINARY_API_KEY = clave privada de Cloudinary

No enviar esas credenciales al HTML.

function cloudinaryConfig_() {
  var properties = PropertiesService.getScriptProperties();
  return {
    cloud: properties.getProperty('CLOUDINARY_CLOUD_NAME'),
    key: properties.getProperty('CLOUDINARY_API_KEY'),
    secret: properties.getProperty('CLOUDINARY_API_SECRET')
  };
}

function cloudinaryFetch_(path, method, payload, contentType) {
  var config = cloudinaryConfig_();
  var auth = Utilities.base64Encode(config.key + ':' + config.secret);
  var options = {
    method: method || 'get',
    headers: { Authorization: 'Basic ' + auth },
    muteHttpExceptions: true
  };
  if (payload !== undefined) {
    options.payload = payload;
    options.contentType = contentType || 'application/json';
  }
  var response = UrlFetchApp.fetch(
    'https://api.cloudinary.com/v1_1/' + config.cloud + path,
    options
  );
  var code = response.getResponseCode();
  var data = JSON.parse(response.getContentText() || '{}');
  if (code < 200 || code >= 300) {
    throw new Error(data.error && data.error.message || 'Cloudinary rechazo la operacion.');
  }
  return data;
}

function mediaList_() {
  var resources = [];
  var nextCursor = null;
  do {
    var body = {
      expression: 'tags=boda-eunice-david',
      max_results: 500,
      sort_by: [{ created_at: 'desc' }]
    };
    if (nextCursor) body.next_cursor = nextCursor;
    var data = cloudinaryFetch_('/resources/search', 'post', JSON.stringify(body));
    (data.resources || []).forEach(function(resource) {
      resources.push({
        public_id: resource.public_id,
        resource_type: resource.resource_type,
        format: resource.format,
        created_at: resource.created_at,
        url: 'https://res.cloudinary.com/' + cloudinaryConfig_().cloud + '/' +
          resource.resource_type + '/upload/' + resource.public_id + '.' + resource.format
      });
    });
    nextCursor = data.next_cursor || null;
  } while (nextCursor);
  return { success: true, resources: resources };
}

function mediaDelete_(rawResources) {
  var resources = JSON.parse(rawResources || '[]');
  if (!Array.isArray(resources) || !resources.length || resources.length > 500) {
    throw new Error('Selecciona entre 1 y 500 archivos.');
  }
  var grouped = { image: [], video: [] };
  resources.forEach(function(resource) {
    if (!resource || typeof resource.public_id !== 'string') return;
    var type = resource.resource_type === 'video' ? 'video' : 'image';
    grouped[type].push(resource.public_id);
  });
  Object.keys(grouped).forEach(function(type) {
    if (!grouped[type].length) return;
    var form = 'public_ids[]=' + grouped[type].map(encodeURIComponent).join('&public_ids[]=');
    cloudinaryFetch_('/resources/' + type + '/upload', 'delete', form, 'application/x-www-form-urlencoded');
  });
  return { success: true, deleted: resources.length };
}
